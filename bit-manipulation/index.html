    count = 0

def divide(dividend: int, divisor: int) -> int:
    if dividend == divisor:
        return 1

    sign = True
    if dividend < 0 and divisor > 0:
        sign = False
    if dividend >= 0 and divisor < 0:
        sign = False

    n = abs(dividend)
    d = abs(divisor)
    quotient = 0
    while n >= d:

        cnt = 0

        while n >= (d << (cnt+1)):
            cnt += 1
        quotient += (1 << cnt)
        n -= (d << cnt)

    if quotient == (1 << 31) and sign:
        return (1 << 31) -1

    if quotient == (1 << 31) and not sign:
        return -(1<<31)        

    return quotient if sign else -quotient

    return 0

def bitManipulation(num : int, i : int) -> List[int]:
    # Write your code here.
    out = []
    ithBith = num & (1 << i-1)
    out.append(1 if ithBith else 0)
    if ithBith == 0:
        num = num | 1 <<i-1
    out.append(num)

    num = num & (~(1<<i-1))
    out.append(num)
    return out

def countPrimes(n: int) -> int:
    if n < 2:
        return 0

    sieve  = [1 for i in range(n)]

    sieve[0] = 0
    sieve[1] = 0

    i = 2
    while i * i < n:

        if sieve[i] == 1:

            for j in range(i*i, n, i):
                sieve[j] = 0
        i += 1
    return sum(sieve)

def main():
    # -------------------------

def sieve(n):

    ans = [1 for i in range(n+1)]

    ans[0] = 0
    ans[1] = 1
    for i in range(2, n+1):
        if ans[i] != 1:
            continue
        ans[i] = i
        for j in range(i*i, n+1, i):
            if ans[j] == 1:
                ans[j] = i
    print(ans)
    return ans

def factors(num, arr):

    ans = []

    while num != 1:
        ans.append(arr[num])
        num = num // arr[num]

    return ans

def main():

def primeFactors(n:int) -> List[int]:

    ans = []
    for i in range(2, int(n**0.5)+1):
        if n % i == 0:
            ans.append(i)

            while n % i == 0:
                n = n // i

    if n != 1:
        ans.append(n)

    return ans

def printDivisors(n: int) -> List[int]:
    # Write your code here
    ans = []
    for i in range(1, int(n**0.5)+1):
        if n % i == 0:
            ans.append(i)
            if n // i != i:
                ans.append(n//i)
    ans.sort()
    return ans

def twoOddNum(arr : List[int]) -> List[int]:
    # Write your code here.
    xor = 0
    for num in arr:
        xor ^= num

    rtsetbit = (xor & (xor-1)) ^ xor

    b1 = 0
    b2 = 0
    for num in arr:

        if num & rtsetbit:
            b1 ^= num
        else:
            b2 ^= num

    return [b1, b2] if b1 > b2 else [b2, b1]

def minBitFlips( start: int, goal: int) -> int:
    cnt = 0
    while start or goal:
        lastbit = start & 1
        endlast = goal & 1
        if lastbit != endlast:
            cnt += 1

        start = start >> 1
        goal = goal >> 1

    return cnt

def minBitFlips1(start: int, goal: int) -> int:
        cnt = 0
        res = start ^ goal
        while res:
            res = res & (res-1)
            cnt += 1

        return cnt

def findXor1toN(n):

    if n % 4 == 1: return 1
    if n % 4 == 2: return n+1
    if n % 4 == 3: return 0

    return n

def findXOR(L : int, R : int) -> int:
    # Write your code here.

    oneToL = findXor1toN(L-1)
    onetoR = findXor1toN(R)

def subsets( nums: List[int]) -> List[List[int]]:
    n = len(nums)
    subsets = 1 << n
    ans = []
    for i in range(subsets):
        temp = []
        for j in range(n):
            if i & (1<<j):
                temp.append(nums[j])
        ans.append(temp)

    return ans
